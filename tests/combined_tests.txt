def fibonacci(n: int) -> list:
    if n <= 0:
        return []
    elif n == 1:
        return [0]
    elif n == 2:
        return [0, 1]
    
    fib_sequence = [0, 1]
    for i in range(2, n):
        next_fib = fib_sequence[-1] + fib_sequence[-2]
        fib_sequence.append(next_fib)
    
    return fib_sequence

    local_storage = StorageLocation(type=StorageType.file, location="/tmp", state_name="test_state.json")

    mng_response = client.post("/tasksmanager", json={"name": "Test TasksManager"})
    assert mng_response.status_code == 201
    mng_id = mng_response.json()["id"]
    
    with open(find_file("tests/calculator/calculator.yaml"), "r") as file:
        swagger_string = file.read()

        "name": ("", "Complex Numbers Calculator API"),
        "description": ("", "Provides Complex Numbers calculator functionalities"),
        "credentials": ("", '{"auth_type":"Baerer","auth_key":"123"}'),

    }

    toolset_response = client.post("/toolsets", files=toolset_data)
    assert toolset_response.status_code == 201
    toolset_id = toolset_response.json()["id"]

    task_response = client.post("/task", json={"name": "Test Task", "toolsets": [toolset_id]})

    task_id = task_response.json()["id"]

    response = client.post("/state/save", json=local_storage.dict())

    # Deserialize from JSON
    response = client.post("/state/load", json={
        "location": local_storage.location, 
        "type": local_storage.type.value,
        "state_name": local_storage.state_name,

    })

    assert loaded_mng_response.status_code == 200
    assert loaded_mng_response.json()['id'] == mng_id

    loaded_toolset_response = client.get(f"/toolsets/{toolset_id}")

from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI()

class ComplexNumber(BaseModel):
    real: float
    imag: float

class ComplexOperation(BaseModel):
    num1: ComplexNumber
    num2: ComplexNumber

def to_complex(num: ComplexNumber):
    return complex(num.real, num.imag)

@app.post("/complex/add")
def add_complex(operation: ComplexOperation):
    num1 = to_complex(operation.num1)
    num2 = to_complex(operation.num2)
    result = num1 + num2
    return {"result": {"real": result.real, "imag": result.imag}}

@app.post("/complex/subtract")
def subtract_complex(operation: ComplexOperation):
    num1 = to_complex(operation.num1)
    num2 = to_complex(operation.num2)
    result = num1 - num2
    return {"result": {"real": result.real, "imag": result.imag}}

@app.post("/complex/multiply")
def multiply_complex(operation: ComplexOperation):
    num1 = to_complex(operation.num1)
    num2 = to_complex(operation.num2)
    result = num1 * num2
    return {"result": {"real": result.real, "imag": result.imag}}

@app.post("/complex/divide")
def divide_complex(operation: ComplexOperation):
    num1 = to_complex(operation.num1)
    num2 = to_complex(operation.num2)
    if num2 == 0:
        return {"error": "Division by zero is not allowed"}
    result = num1 / num2
    return {"result": {"real": result.real, "imag": result.imag}}

from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI()


class Operation(BaseModel):
    num1: float
    num2: float


@app.post("/add")
def add(operation: Operation):
    return {"result": operation.num1 + operation.num2}


@app.post("/subtract")
def subtract(operation: Operation):
    return {"result": operation.num1 - operation.num2}


@app.post("/multiply")
def multiply(operation: Operation):
    return {"result": operation.num1 * operation.num2}


@app.post("/divide")
def divide(operation: Operation):
    if operation.num2 == 0:
        return {"error": "Division by zero is not allowed"}
    return {"result": operation.num1 / operation.num2}
