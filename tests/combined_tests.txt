def  fibonacci(n: int) -> list:
    if n <= 0:
        return []
    elif n == 1:
        return [0]
    elif n == 2:
        return [0, 1]

    fib_sequence = [0, 1]
    for i in range(2, n):
        next_fib = fib_sequence[-1] + fib_sequence[-2]
        fib_sequence.append(next_fib)

    return fib_sequence

import requests
from requests.auth import HTTPBasicAuth
import json
from typing import List, Dict

class JiraService:
    def __init__(self, jira_url: str, username: str, api_token: str):
        self.jira_url = jira_url
        self.auth = HTTPBasicAuth(username, api_token)
        self.headers = {
            'Accept': 'application/json',
            'Content-Type': 'application/json'
        }

    def fetch_tickets(self, project_key: str, max_results: int = 50) -> List[Dict]:
        jql_query = f'project = {project_key} ORDER BY created DESC'
        url = f'{self.jira_url}/rest/api/2/search'
        payload = {
            'jql': jql_query,
            'maxResults': max_results,
            'fields': ['summary', 'status', 'assignee', 'created', 'updated']
        }
        response = requests.post(url, headers=self.headers, auth=self.auth, json=payload)
        response.raise_for_status()
        return response.json().get('issues', [])

    def update_ticket_status(self, ticket_id: str, status: str) -> bool:
        url = f'{self.jira_url}/rest/api/2/issue/{ticket_id}/transitions'
        transitions_response = requests.get(url, headers=self.headers, auth=self.auth)
        transitions_response.raise_for_status()
        transitions = transitions_response.json()['transitions']
        
        for transition in transitions:
            if transition['to']['name'].lower() == status.lower():
                transition_id = transition['id']
                payload = {'transition': {'id': transition_id}}
                response = requests.post(url, headers=self.headers, auth=self.auth, json=payload)
                response.raise_for_status()
                return True
        return False

from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI()

class ComplexNumber(BaseModel):
    real: float
    imag: float

class ComplexOperation(BaseModel):
    num1: ComplexNumber
    num2: ComplexNumber

def to_complex(num: ComplexNumber):
    return complex(num.real, num.imag)

@app.post("/complex/add")
def add_complex(operation: ComplexOperation):
    num1 = to_complex(operation.num1)
    num2 = to_complex(operation.num2)
    result = num1 + num2
    return {"result": {"real": result.real, "imag": result.imag}}

@app.post("/complex/subtract")
def subtract_complex(operation: ComplexOperation):
    num1 = to_complex(operation.num1)
    num2 = to_complex(operation.num2)
    result = num1 - num2
    return {"result": {"real": result.real, "imag": result.imag}}

@app.post("/complex/multiply")
def multiply_complex(operation: ComplexOperation):
    num1 = to_complex(operation.num1)
    num2 = to_complex(operation.num2)
    result = num1 * num2
    return {"result": {"real": result.real, "imag": result.imag}}

@app.post("/complex/divide")
def divide_complex(operation: ComplexOperation):
    num1 = to_complex(operation.num1)
    num2 = to_complex(operation.num2)
    if num2 == 0:
        return {"error": "Division by zero is not allowed"}
    result = num1 / num2
    return {"result": {"real": result.real, "imag": result.imag}}

from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI()


class Operation(BaseModel):
    num1: float
    num2: float


@app.post("/add")
def add(operation: Operation):
    return {"result": operation.num1 + operation.num2}


@app.post("/subtract")
def subtract(operation: Operation):
    return {"result": operation.num1 - operation.num2}


@app.post("/multiply")
def multiply(operation: Operation):
    return {"result": operation.num1 * operation.num2}


@app.post("/divide")
def divide(operation: Operation):
    if operation.num2 == 0:
        return {"error": "Division by zero is not allowed"}
    return {"result": operation.num1 / operation.num2}
