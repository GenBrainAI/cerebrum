from typing import Optional
from github import Github

class GithubTool:
    def __init__(self, access_token: str, repo_name: str):
        if access_token is None:
            raise ValueError("Access token cannot be None")
        if repo_name is None:
            raise ValueError("Repository name cannot be None")

        try:
            self.g = Github(access_token)
            self.repo = self.g.get_repo(repo_name)
        except Exception as e:
            print(f"Failed to get repository: {e}")
            self.repo = None

        if self.repo is None:
            raise Exception(f"Could not access repository {repo_name} with provided token.")

    async def ls(self, path: str = '') -> list:
        return self.repo.ls(path)

    async def get_readme(self, path: str = '') -> Optional[str]:
        return self.repo.get_readme(path)

    async def read_file(self, file_path: str) -> Optional[str]:
        return self.repo.read_file(file_path)

    async def create_branch(self, branch_name: str) -> str:

    async def commit_file_changes(self, file_path: str, commit_message: str, file_content: str) -> Optional[str]:

def save_string_to_file(filename: str, content: str, base_path: str = "."):
    # Ensure the directory exists
    os.makedirs(base_path, exist_ok=True)
    
    # Construct the full path
    file_path = os.path.join(base_path, filename)
    
    # Write the content to the file
    with open(file_path, "w") as file:
        file.write(content)

def run_file(filename: str, language: str, base_path: str = "."):
    # Construct the full path
    file_path = os.path.join(base_path, filename)
    
    # Determine the command based on the language
    if language == "python":
        command = ["python", file_path]
    elif language == "nodejs":
        command = ["node", file_path]
    else:
        raise ValueError("Unsupported language")

    # Run the command and capture the output
    result = subprocess.run(command, capture_output=True, text=True)
    
    # Check if the command was successful
    if result.returncode != 0:
        raise RuntimeError(result.stderr)
    
    return result.stdout


class ChromaSingleton:
    _instance = None
    _db = None
    _persist_directory = "./chroma_db"

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super(ChromaSingleton, cls).__new__(cls)
            cls._instance._init_db()
        return cls._instance

    def _init_db(self):
        embedding_function = OpenAIEmbeddings(api_key=OPENAI_API_KEY)
        if os.path.exists(self._persist_directory):
            self._db = Chroma(persist_directory=self._persist_directory, embedding_function=embedding_function)
        else:
            self._db = None

    def get_db(self):
        return self._db

    def set_db(self, db):
        self._db = db


@app.get("/vector/load_self_code")
async def load_cerebrum_code():
    try:
        # Ensure singleton instance
        chroma_singleton = ChromaSingleton()

        # Find all relevant files in the project root
        project_root = "."
        file_extensions = ['*.py', '*.yml', '*.yaml', '*.md', 'requirements.txt', 'Dockerfile', '.env']
        project_files = []
        for ext in file_extensions:
            project_files.extend([y for x in os.walk(project_root) for y in glob(os.path.join(x[0], ext))])

        # Load the content of all files
        documents = []
        for file_path in project_files:
            loader = TextLoader(file_path)
            documents.extend(loader.load())

        # Split documents into chunks
        text_splitter = CharacterTextSplitter(chunk_size=1000, chunk_overlap=0)
        docs = text_splitter.split_documents(documents)

        # Create the embedding function
        embedding_function = OpenAIEmbeddings(api_key=OPENAI_API_KEY)

        # Load documents into Chroma
        db = Chroma.from_documents(docs, embedding_function, persist_directory=chroma_singleton._persist_directory)

        # Update the singleton instance
        chroma_singleton.set_db(db)

        return {"message": "Successfully loaded code into Chroma database"}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

def create_toolset(credentials, name, description, swagger_content):
    credentials_dict = json.loads(credentials)
    toolset_credentials = ToolsetCredentials(**credentials_dict)

    toolset_data = {
        "name": name,
        "description": description,
        "credentials": toolset_credentials,
        "swagger": swagger_content
    }

    toolset = Toolset(**toolset_data)
    persisted_toolset = persistence_model.toolsets.create(toolset)
    parsed_tools = openapi.parse_openapi(persisted_toolset.swagger)

    persisted_toolset.tool_ids = []
    for parsed_tool in parsed_tools:
        try:
            function_data = parsed_tool['function']
            
            # Convert parameters
            parameters = {}
            for param_type, param_value in function_data['parameters'].items():
                if param_type == 'requestBody':
                    parameters[param_type] = ResolvedSchema.parse_obj(param_value)
                else:
                    parameters[param_type] = {
                        name: ParameterSchema(name=name, in_=param_type, **param_info)
                        for name, param_info in param_value.items()
                    }

            # Convert responses
            responses = {
                status: ResponseContent(
                    description=resp_data['description'],
                    content={
                        content_type: {
                            'schema': ResolvedSchema.parse_obj(content_schema)
                        }
                        for content_type, content_schema in resp_data['content'].items()
                    }
                )
                for status, resp_data in function_data['responses'].items()
            }

            function = Function(
                name=function_data['name'],
                url=function_data['url'],
                method=function_data['method'],
                description=function_data['description'],
                parameters=parameters,
                responses=responses,
                info=Info(**function_data['info']),
                servers=[ServerInfo(**s) for s in function_data['servers']],
                credentials=toolset_credentials,
                toolsetId=persisted_toolset.id
            )

            new_tool = persistence_model.tools.create(Tool(function=function, type="function"))
            persisted_toolset.tool_ids.append(new_tool.id)
        except Exception as e:
            print(f"Error creating tool: {str(e)}")

    return persisted_toolset

    tokens = 0

# get messages the can fit to the model limitation,
# this function iterate over the messages backwards
# calculate the length of each message, until the sum of tokens
# reaches the model limitation, then return the messages
# without the last one (keeping 1000 tokens for the response)
# the function also checks for duplicated messages and exclude them
def get_messages_per_model_limit(model: str, message_history: MongoDBChatMessageHistory):
    max_model_tokens = max_tokens(model) - 1000  # Reserve 1000 tokens for the response
    encoding = tiktoken.encoding_for_model(model)

    total_tokens = 0
    unique_message_hashes = set()
    included_messages = []

    # Function to create a hash of a message content
    def hash_content(content):
        return hashlib.sha256(content.encode()).hexdigest()

    # Iterate over the messages in reverse order
    for message in reversed(message_history.messages):
        message_hash = hash_content(message.content)

        # Skip duplicate messages based on hash
        if message_hash in unique_message_hashes:
            continue
        unique_message_hashes.add(message_hash)

        message_token_count = len(encoding.encode(message.content))
        total_tokens += message_token_count

        # If adding this message would exceed the token limit, break the loop
        if total_tokens > max_model_tokens:
            break

        included_messages.append(message)

    # Reverse the message list, so it's in the original order
    return list(reversed(included_messages))


# In case a message is too long, we can summarise it, to shorter version
# we should check the length of the message, select the model that can handle it
# and then summarise it to fit the max tokens of the given model
# we should use the following example
#     messages = [{"role": "user", "content": "please summarise this message, it is too long"}]
#     client = OpenAI(api_key=os.environ.get("OPENAI_API_KEY"))
#     response = client.chat.completions.create(
#         model='gpt-3.5-turbo-0613',
#         messages=messages,
#     )
#
#   in case the response is too long, we should summarise it again, until it fits the model by given factor
def summarise_response(model: str, message: str, factor: float = 0.5):
    # Ensure the factor is between 0 and 1
    factor = max(0., min(factor, 1))

    # Check and trim the original message if needed
    trimmed_message = check_and_trim_message(message)

    # Calculate the desired maximum length of the summary
    max_length = int(len(trimmed_message) * factor)

    # Function to generate a summary
    def generate_summary(content):
        try:
            response = client.chat.completions.create(
                model=model,
                messages=[
                    {"role": "system", "content": "Please summarize the following content."},
                    {"role": "user", "content": content}
                ]
            )
            return response.choices[0].message['content'].strip()
        except Exception as e:
            raise HTTPException(status_code=500, detail=str(e))

    # Initial summary generation
    summary = generate_summary(trimmed_message)

    # If the summary is still too long, repeat the summarization
    while len(summary) > max_length:
        summary = generate_summary(summary)

    # Final check if the summary fits the model's token limit
    final_summary = check_and_trim_message(summary)

    return final_summary


def check_and_trim_message(content, encoding, max_model_tokens):
    # Calculate the current number of tokens in the message
    current_token_count = len(encoding.encode(content))

    # If the current token count is within the model's limit, return the original content
    if current_token_count <= max_model_tokens:
        return content

    # Calculate the ratio of max tokens to current tokens
    trim_ratio = max_model_tokens / current_token_count

    # Apply a threshold to avoid trimming too much in case of slightly exceeding the limit
    trim_ratio *= 0.90

    # Calculate the approximate character count to keep
    char_count_to_keep = int(len(content) * trim_ratio)

    # Trim the message
    return content[:char_count_to_keep]

            tool = persistence_model.tools.read(tid)
            if tool['type'] != 'function':
                continue
            function = tool['function']
            fname = f"fn{index}_{function['name']}"
            functions_dict[fname] = function
            functions_dict[fname]["tool_id"] = tid
    if len(functions_dict) > 0:
        tools = transform_functions_for_openai(functions_dict)
        return tools, functions_dict
    else:
        return [], {}


def transform_functions_for_openai(functions_dict):
    """
    Transforms function definitions into a schema suitable for OpenAI tools.

    :param functions_dict: A dictionary of functions to transform.
    :return: A list of transformed function definitions.
    """
    transformed_functions = []
    for fname, function in functions_dict.items():
        simplified_params = {
            'type': 'object',
            'properties': {},
            'required': []}

        if 'requestBody' in function['parameters']:
            request_body = function['parameters']['requestBody']
            request_body_json = json.dumps(request_body)
            print(f'request_body_json: {request_body_json}')
            simplified_params['properties']['requestBody'] = transform_schema(request_body)
            simplified_params_json = json.dumps(simplified_params)
            print(f'simplified_params_json: {simplified_params_json}')

        for param_type in ['query', 'header', 'path']:
            for param_name, param_info in function['parameters'].get(param_type, {}).items():
                simplified_params['properties'][param_name] = transform_schema(param_info)
                if param_info.get('required', False):
                    simplified_params['required'].append(param_name)

        transformed_functions.append({
            "type": "function",
            "function": {
                'name': fname,
                'description': function.get('description', ''),
                'parameters': simplified_params,
            }
        })

    return transformed_functions


def transform_schema(schema):
    """
    Recursively transforms a schema into a simpler, OpenAI-compatible format.

    :param schema: The schema to transform.
    :return: The transformed schema.
    """
    if schema.get('type') == 'object' and 'properties' in schema:
        transformed_properties = {}
        for prop_name, prop_info in schema['properties'].items():
            transformed_properties[prop_name] = transform_schema(prop_info)
        return {
            'type': 'object',
            'properties': transformed_properties,
            'required': schema.get('required', [])
        }
    elif schema.get('type') == 'array' and 'items' in schema:
        return {
            'type': 'array',
            'items': transform_schema(schema['items'])
        }
    else:
        return {
            'type': schema.get('type', 'string'),
            'description': schema.get('description', '')
        }



def extract_code_segment(text, mode):
    """
    Extracts code data from a string based on specified code type within markers like ```code_type ... ```.

    Parameters:
    - text (str): A string containing code data encapsulated within ```code_type ... ``` markers.
    - code_type (str): The type of code to extract ('yaml', 'json', 'python', 'xml', 'csv', 'markdown').

    Returns:
    - Depending on code_type:
        - dict, list, or str if YAML, JSON, Python, XML, or Markdown.
        - list of dicts if CSV.
    - None: If extraction or decoding fails, or if the code type is not recognized.
    """
    try:
        start_marker = f"```{mode}"
        end_marker = "```"
        start_index = text.find(start_marker) + len(start_marker)
        if start_index == -1 + len(start_marker):
            return None  # Start marker not found

        end_index = text.find(end_marker, start_index)
        if end_index == -1:
            return None  # End marker not found

        code_str = text[start_index:end_index].strip()

        if mode == 'yaml':
            try:
                return yaml.safe_load(code_str)
            except yaml.YAMLError:
                return None
        elif mode == 'json':
            try:
                return json.loads(code_str)
            except json.JSONDecodeError:
                try:
                    return json.loads("{" + code_str + "}")
                except json.JSONDecodeError:
                    return None
        elif mode == 'python':
            return code_str  # Returning Python code as string; execution or further processing is not handled here.
        elif mode == 'xml':
            # TOFDO: Fix this
            # try:
            #     return ET.fromstring(code_str)  # Returns an ElementTree element
            # except ET.ParseError:
            return None
        elif mode == 'csv':
            try:
                reader = csv.DictReader(io.StringIO(code_str))
                return list(reader)  # Returns list of dictionaries
            except csv.Error:
                return None
        elif mode == 'markdown':
            return code_str  # Markdown is returned as a string; rendering is not handled here.
        else:
            return None  # Code type not supported
    except Exception as e:
        ExtractModeError(mode=mode, response=text)


def find_file(file_base_loc: str, limit: int = 2):
    output_path = file_base_loc
    for i in range(0, limit):
        if os.path.isfile(output_path):
            return output_path
        else:
            output_path = os.path.join("..", output_path)
    raise Exception(f"{file_base_loc} not found")

class TaskAggregator(ABC):
    @abstractmethod
    def aggregate(self, responses: List[str]) -> str:
        pass

class AllAggregator(TaskAggregator):
    def aggregate(self, responses: List[str]) -> str:
        return "\n".join(responses)

class CombineAggregator(TaskAggregator):
    def aggregate(self, responses: List[str]) -> str:
        return " ".join(responses)

class SummarizeAllAggregator(TaskAggregator):
    def aggregate(self, responses: List[str]) -> str:
        # Simulating summarization logic
        return f"Summary: {', '.join(responses)}"

class ChooseFunctionAggregator(TaskAggregator):
    def __init__(self, choose_function: Callable[[List[str]], str]):
        self.choose_function = choose_function

    def aggregate(self, responses: List[str]) -> str:
        return self.choose_function(responses)

class TaskNode:
    def __init__(self, task: Task, aggregator: TaskAggregator = AllAggregator()):
        self.task = task
        self.aggregator = aggregator
        self.children: List[TaskNode] = []

    def add_child(self, child: 'TaskNode'):
        self.children.append(child)

    def run(self, additional_info: Optional[Dict] = None) -> str:
        responses = []
        for child in self.children:
            response = child.run(additional_info)
            responses.append(response)

        aggregated_response = self.aggregator.aggregate(responses)
        return self.task.run({"input_data": aggregated_response})

class TasksManager:
    def __init__(self, root: TaskNode):
        self.root = root

    def run(self, additional_info: Optional[Dict] = None) -> str:
        return self.root.run(additional_info)

from functools import partial
import inspect
import httpx

class PythonFunctionTool(BaseModel):
    name: str
    function: Callable
    description: str = ""

class CurriedProxy(BaseModel):
    id: UUID = Field(default_factory=uuid4)
    tools_dict: Dict[str, Union[Tool, PythonFunctionTool]] = Field(default_factory=dict)
    name_counter: Dict[str, int] = Field(default_factory=dict)

    class Config:
        arbitrary_types_allowed = True

    def generate_safe_name(self, original_name: str) -> str:
        # Remove non-alphanumeric characters and replace spaces with underscores
        safe_name = re.sub(r'[^\w\s-]', '', original_name).strip().replace(' ', '_')
        
        # Ensure the name starts with a letter or underscore
        if not safe_name[0].isalpha() and safe_name[0] != '_':
            safe_name = f"tool_{safe_name}"
        
        # If the name already exists, append a number
        base_name = safe_name
        if base_name in self.name_counter:
            self.name_counter[base_name] += 1
            safe_name = f"{base_name}_{self.name_counter[base_name]}"
        else:
            self.name_counter[base_name] = 0
        
        return safe_name


    async def multiset_proxy(self, name: str, arguments: Dict[str, Any]) -> Any:
        tool = self.tools_dict[name]
        try:
            if isinstance(tool, PythonFunctionTool):
                return await self._call_python_function(tool.function, arguments)
            elif isinstance(tool, Tool):
                return await self._call_api_function(tool, arguments)
            else:
                raise ValueError(f"Unknown tool type for {name}")
        except Exception as e:
            return {"error": str(e)}

    async def _call_python_function(self, func: Callable, arguments: Dict[str, Any]) -> Any:
        if inspect.iscoroutinefunction(func):
            return await func(**arguments)
        else:
            return func(**arguments)

    async def _call_api_function(self, tool: Tool, arguments: Dict[str, Any]) -> Dict[str, Any]:
        function: Function = tool.function
        base_url = function.servers[0].url
        url = base_url + function.url
        headers = {}
        credentials: ToolsetCredentials = function.credentials

        if credentials.auth_type and credentials.auth_key:
            headers["Authorization"] = f"{credentials.auth_type}: {credentials.auth_key}"

        query_params = {}
        header_params = {}
        path_params = {}
        body_params = {}

        if credentials.path_credentials:
            for key_set in credentials.path_credentials:
                path_params[key_set.key] = key_set.value

        for param_type, params in function.parameters.items():
            if param_type == 'query':
                query_params = {k: arguments.get(k) for k in params if k in arguments}
            elif param_type == 'header':
                header_params = {k: arguments.get(k) for k in params if k in arguments}
            elif param_type == 'path':
                path_params.update({k: arguments.get(k) for k in params if k in arguments})
            elif param_type == 'requestBody':
                body_params = arguments.get('requestBody', {})

        headers.update(header_params)
        url = url.format(**path_params)

        headers["Content-Type"] = "application/json" if body_params else "application/x-www-form-urlencoded"
        headers["Accept"] = "*/*"

        async with httpx.AsyncClient() as client:
            method = function.method.lower()
            request_kwargs = {
                "url": url,
                "params": query_params,
                "headers": headers,
            }
            if method in ['post', 'put', 'patch']:
                request_kwargs["json" if headers["Content-Type"] == "application/json" else "data"] = body_params

            response = await getattr(client, method)(**request_kwargs)

        return response.json()

    def curried_proxy(self, name: str):
        return partial(self.multiset_proxy, name)

    def add_tool(self, name: str, tool: Union[Tool, Callable, object]):
        safe_name = self.generate_safe_name(name)
        if isinstance(tool, Tool):
            self.tools_dict[safe_name] = tool
        elif callable(tool):
            self.tools_dict[safe_name] = PythonFunctionTool(name=safe_name, function=tool, description=tool.__doc__ or "")
        elif isinstance(tool, object):
            for method_name, method in inspect.getmembers(tool, predicate=inspect.ismethod):
                if not method_name.startswith('_'):
                    full_name = self.generate_safe_name(f"{name}.{method_name}")
                    self.tools_dict[full_name] = PythonFunctionTool(
                        name=full_name,
                        function=method,
                        description=method.__doc__ or ""
                    )
        else:
            raise ValueError(f"Unsupported tool type: {type(tool)}")

    def remove_tool(self, name: str):
        self.tools_dict.pop(name, None)

    def get_tool(self, name: str) -> Union[Tool, PythonFunctionTool]:
        return self.tools_dict.get(name)

    def list_tools(self) -> List[str]:
        return list(self.tools_dict.keys())

    def create_crewai_tool(self, name: str, tool: Union[Tool, PythonFunctionTool]) -> BaseTool:
        tool_description = tool.function.description if isinstance(tool, Tool) else tool.description

        class CustomTool(BaseTool):
            name: str
            description: str
            proxy: CurriedProxy

            def _run(self, **kwargs: Any) -> Any:
                return self.proxy.curried_proxy(self.name)(kwargs)

        custom_tool = CustomTool(
            name=name,
            description=tool_description,
            proxy=self
        )
        return custom_tool

    def export_to_crewai(self) -> List[BaseTool]:
        crewai_tools = []
        for name, tool in self.tools_dict.items():
            crewai_tool = self.create_crewai_tool(name, tool)
            crewai_tools.append(crewai_tool)
        return crewai_tools

    def export_to_openai(self) -> List[Dict[str, Any]]:
        openai_tools = []
        for name, tool in self.tools_dict.items():
            if isinstance(tool, Tool):
                function = tool.function
                openai_tool = {
                    "type": "function",
                    "function": {
                        "name": name,
                        "description": function.description,
                        "parameters": self._convert_parameters_to_openai_schema(function.parameters)
                    }
                }
                openai_tools.append(openai_tool)
            elif isinstance(tool, PythonFunctionTool):
                openai_tool = {
                    "type": "function",
                    "function": {
                        "name": name,
                        "description": tool.description,
                        "parameters": self._get_function_parameters(tool.function)
                    }
                }
                openai_tools.append(openai_tool)
        return openai_tools

    def export_to_anthropic(self) -> List[Dict[str, Any]]:
        anthropic_tools = []
        for name, tool in self.tools_dict.items():
            if isinstance(tool, Tool):
                function = tool.function
                anthropic_tool = {
                    "type": "function",
                    "name": name,
                    "description": function.description,
                    "parameters": self._convert_parameters_to_anthropic_schema(function.parameters)
                }
                anthropic_tools.append(anthropic_tool)
            elif isinstance(tool, PythonFunctionTool):
                anthropic_tool = {
                    "type": "function",
                    "name": name,
                    "description": tool.description,
                    "parameters": self._get_function_parameters(tool.function)
                }
                anthropic_tools.append(anthropic_tool)
        return anthropic_tools

    def _convert_parameters_to_openai_schema(self, parameters: Dict[str, Any]) -> Dict[str, Any]:
        schema = {
            "type": "object",
            "properties": {},
            "required": []
        }
        for param_type, params in parameters.items():
            if param_type == 'requestBody':
                schema["properties"]["requestBody"] = self._convert_schema(params)
            else:
                for name, param in params.items():
                    schema["properties"][name] = self._convert_schema(param.schema)
                    if param.required:
                        schema["required"].append(name)
        return schema

    def _convert_parameters_to_anthropic_schema(self, parameters: Dict[str, Any]) -> Dict[str, Any]:
        # Anthropic's schema is similar to OpenAI's, so we can reuse the conversion
        return self._convert_parameters_to_openai_schema(parameters)

    def _convert_schema(self, schema: Union[Dict[str, Any], ResolvedSchema]) -> Dict[str, Any]:
        if isinstance(schema, dict):
            converted = {"type": schema.get("type", "string")}
            if "description" in schema:
                converted["description"] = schema["description"]
            if "properties" in schema:
                converted["properties"] = {
                    k: self._convert_schema(v) for k, v in schema["properties"].items()
                }
            if "items" in schema:
                converted["items"] = self._convert_schema(schema["items"])
        elif isinstance(schema, ResolvedSchema):
            converted = {"type": schema.type or "string"}
            if schema.description:
                converted["description"] = schema.description
            if schema.properties:
                converted["properties"] = {
                    k: self._convert_schema(v) for k, v in schema.properties.items()
                }
            if schema.items:
                converted["items"] = self._convert_schema(schema.items)
        else:
            # If it's neither a dict nor a ResolvedSchema, return a default schema
            converted = {"type": "string"}
        
        return converted

    def _get_function_parameters(self, func: Callable) -> Dict[str, Any]:
        sig = inspect.signature(func)
        parameters = {
            "type": "object",
            "properties": {},
            "required": []
        }
        for name, param in sig.parameters.items():
            if param.annotation != inspect.Parameter.empty:
                parameters["properties"][name] = {"type": str(param.annotation.__name__)}
            else:
                parameters["properties"][name] = {"type": "string"}
            if param.default == inspect.Parameter.empty:
                parameters["required"].append(name)
        return parameters

def generate_function_name(method, path, method_content):
    """
    Generate a function name based on the HTTP method and path in method_content.
    Checks for 'operationId' or 'name' first, and if neither is present, generates a name.
    """
    # Check for 'operationId' or 'name'
    if 'name' in method_content:
        return method_content['name']

    if 'operationId' in method_content:
        return method_content['operationId']

    #print(f'return method: {method} path: {path}')

    # Split the path and remove empty strings
    path_parts = [p for p in path.split('/') if p]
    # Replace path parameters with general terms (e.g., {id} -> ById)
    name_parts = [p.replace('{', '').replace('}', '') if '{' in p else p.capitalize() for p in path_parts]
    # Join the parts with CamelCase or snake_case format
    function_name = ''.join(name_parts)
    name = method.lower() + function_name.capitalize()
    return name


def resolve_schema(schema, parser):
    if '$ref' in schema:
        ref_path = schema['$ref'].split('/')
        ref = parser.specification
        for path in ref_path[1:]:
            ref = ref[path]
        return resolve_schema(ref, parser)
    
    resolved_schema = {}
    jdump = json.dumps(schema)
    if 'type' in schema:
        resolved_schema['type'] = schema['type']
    if 'properties' in schema:
        resolved_schema['properties'] = {}
        for prop_name, prop_attrs in schema['properties'].items():
            resolved_schema['properties'][prop_name] = resolve_schema(prop_attrs, parser)
    if 'items' in schema:
        resolved_schema['items'] = resolve_schema(schema['items'], parser)
    
    return resolved_schema

def parse_openapi(swagger_yaml):
    tools = []
    try:
        parser = ResolvingParser(spec_string=swagger_yaml)
        info = parser.specification['info']
        servers = parser.specification['servers']
        
        for path, path_content in parser.specification['paths'].items():
            for method, method_content in path_content.items():
                if method in ['get', 'put', 'post', 'delete', 'options', 'head', 'patch', 'trace']:
                    name = generate_function_name(method, path, method_content)
                    tool = {
                        "type": "function",
                        "function": {
                            'name': name,
                            'url': path,
                            'method': method,
                            'description': method_content.get('summary', method_content.get('description', '')),
                            'parameters': {},
                            'responses': {},
                            'info': info,
                            'servers': servers
                        }
                    }
                    
                    request_body = method_content.get('requestBody', {})
                    if request_body:
                        content = request_body.get('content', {})
                        for content_type, content_value in content.items():
                            schema = content_value.get('schema', {})
                            resolved_schema = resolve_schema(schema, parser)
                            tool['function']['parameters']['requestBody'] = resolved_schema
                    
                    for param in method_content.get('parameters', []):
                        if 'name' in param and 'schema' in param:
                            param_type = param['in']
                            if param_type not in tool['function']['parameters']:
                                tool['function']['parameters'][param_type] = {}
                            param_info = resolve_schema(param['schema'], parser)
                            param_info['description'] = param.get('description', '')
                            param_info['required'] = param.get('required', False)
                            tool['function']['parameters'][param_type][param['name']] = param_info
                    
                    responses = method_content.get('responses', {})
                    for status_code, response_content in responses.items():
                        if 'content' in response_content:
                            for content_type, content_value in response_content['content'].items():
                                schema = content_value.get('schema', {})
                                resolved_schema = resolve_schema(schema, parser)
                                tool['function']['responses'][status_code] = {
                                    'description': response_content.get('description', ''),
                                    'content': {
                                        content_type: resolved_schema
                                    }
                                }
                    tool_json = json.dumps(tool)
                    # print(f'tool_json: {tool_json}')
                    tools.append(tool)
    except Exception as e:
        print(str(e))
        raise HTTPException(status_code=404, detail=str(e))

    return tools

        messages = self._messages_to_openai(messages)

        while True:
            response = await self.client.chat.completions.create(
                model=model,
                messages=messages,
                tools=tools if tools else None,
                tool_choice="auto" if tools else None
            ) 
            # if tools else await self.client.chat.completions.create(
            #     model=model,
            #     messages=messages
            # )
            choice = response.choices[0]
            chat_completion_message = choice.message.content

            if choice.finish_reason in ["stop", "length"]:
                # Add the final message to the history and return it
                messages.append({'role': 'assistant', 'content': chat_completion_message})
                return chat_completion_message
            elif choice.finish_reason == "content_filter":
                # Handle content filter cases
                filtered_message = "Content filtered by model: " + chat_completion_message
                messages.append({'role': 'assistant', 'content': chat_completion_message})
                return filtered_message
            elif choice.finish_reason == "tool_calls":
                tool_calls = choice.message.tool_calls
                messages.append({'role': response.choices[0].message.role, 'tool_calls': tool_calls})
                # Prepare coroutines for all function calls
                coroutine_list = []
                tool_calls = choice.message.tool_calls
                for tool_call in tool_calls:
                    function_name = tool_call.function.name
                    function_args = json.loads(tool_call.function.arguments)
                    coroutine = curried_proxy.curried_proxy(function_name)(function_args)
                    coroutine_list.append(coroutine)

                # Run all coroutines concurrently and wait for all responses
                responses = await asyncio.gather(*coroutine_list)
                # Process each response and add it to messages
                for tool_call, function_response in zip(tool_calls, responses):
                    if isinstance(function_response, dict) or isinstance(function_response, list):
                        response_str = json.dumps(function_response)
                    else:
                        response_str = str(function_response)

                    messages.append({
                        "role": "tool",
                        "name": tool_call.function.name,
                        "tool_call_id": tool_call.id,
                        "content": response_str
                    })
            else:
                raise ValueError(f"Unexpected finish reason: {choice.finish_reason}")

    async def _anthropic_completion(self, model, messages, curried_proxy: CurriedProxy):

            anthropic_messages = self._messages_to_anthropic(messages)

                "max_tokens": 1000,  # You can adjust this value as needed
                "messages": anthropic_messages,
            }

            # Only add the tools parameter if there are tools
            if tools:

            response = await self.client.messages.create(**api_params)

            else:
                raise ValueError("No content in Anthropic response")

                }

    async def _langchain_completion(self, model: str, messages: List[Dict[str, Any]], curried_proxy: CurriedProxy) -> str:
        try:
            tools = curried_proxy.export_to_openai()  # We'll use OpenAI function format for compatibility
            langchain_messages = self._convert_messages_to_langchain(messages)

            # Create the appropriate LangChain chat model based on the model name
            if model.startswith("gpt-"):
                llm = ChatOpenAI(model_name=model, temperature=0)
            elif model.startswith("claude-"):
                llm = ChatAnthropic(model=model)
            else:
                raise ValueError(f"Unsupported model for LangChain: {model}")

            # Create a ChatPromptTemplate from the messages
            prompt = ChatPromptTemplate.from_messages(langchain_messages)

            # Create an LLMChain
            chain = LLMChain(llm=llm, prompt=prompt)

            while True:
                # Execute the chain
                response = await chain.arun(tools=tools)

                # Check if the response contains function calls
                if "function_call" in response:
                    function_call = json.loads(response["function_call"])
                    function_name = function_call["name"]
                    function_args = function_call["arguments"]

                    # Execute the function
                    result = await curried_proxy.curried_proxy(function_name)(function_args)

                    # Add the function result to the messages
                    langchain_messages.append(FunctionMessage(content=json.dumps(result), name=function_name))

                    # Continue the conversation with the function result
                    continue

                # If no function call, return the response
                return response
        except Exception as e:
            logging.error(f"Error in LangChain completion: {e}")
            raise HTTPException(status_code=500, detail=str(e))

        
    def _convert_messages_to_langchain(self, messages: List[Any]) -> List[Any]:
        langchain_messages = []
        for message in messages:
            if message.role == "system":
                langchain_messages.append(SystemMessage(content=message.content))
            elif message.role == "user":
                langchain_messages.append(HumanMessage(content=message.content))
            elif message.role == "assistant":
                langchain_messages.append(AIMessage(content=message.content))
            elif message.role == "function":
                langchain_messages.append(FunctionMessage(content=message.content, name=message.name))
        return langchain_messages

import httpx

class RestProxyController:

    async def multiset_proxy(self, name: str, arguments: Dict[str, Any], tools_dict: Dict[str, Tool]):
        tool = tools_dict[name]
        function: Function = tool.function
        
        # Prepare URL
        base_url = function.servers[0].url
        url = base_url + function.url
        headers = {}
        credentials: ToolsetCredentials = function.credentials
        
        # Prepare headers with Authentication token
        if credentials.auth_type and credentials.auth_key:
            headers = {
                "Authorization": f"{credentials.auth_type}: {credentials.auth_key}"
            }

        # Organize parameters by type
        query_params = {}
        header_params = {}
        path_params = {}
        body_params = {}

        # Check for additional global and/or security keys in toolset and add them to path_params
        if credentials.path_credentials:
            for key_set in credentials.path_credentials:
                path_params[key_set.key] = key_set.value

        for param_type, params in function.parameters.items():
            if param_type == 'query':
                query_params = {k: arguments[k] for k, v in params.items() if k in arguments}
            elif param_type == 'header':
                header_params = {k: arguments[k] for k, v in params.items() if k in arguments}
            elif param_type == 'path':
                path_params.update({k: arguments[k] for k, v in params.items() if k in arguments})
            elif param_type == 'requestBody':
                body_params = arguments.get('requestBody', {})

        # Update headers and URL with path and header parameters
        headers.update(header_params)
        url = url.format(**path_params)  # Assuming path parameters are used as {param} in the URL

        # Choose the appropriate content type
        if body_params:
            headers["Content-Type"] = "application/json"

        headers["Accept"] = "*/*"

        return await self._send_request(function.method, url, query_params, body_params, headers)

    async def proxy(self, function_name: str, event: Dict[str, Any], tool: Tool):
        function: Function = tool.function
        
        # Prepare URL
        base_url = function.servers[0].url
        url = base_url + function.url

        # Organize parameters by type
        headers = event.get('headers', {})
        query_params = event.get('queryStringParameters', {})
        path_params = event.get('pathParameters', {})
        body_params = event.get('body', {})

        credentials: ToolsetCredentials = function.credentials
        # Prepare headers with Authentication token
        if credentials.auth_type and credentials.auth_key:
            headers["Authorization"] = f"{credentials.auth_type}: {credentials.auth_key}"

        # Check for additional global and/or security keys in toolset and add them to path_params
        if credentials.path_credentials:
            for key_set in credentials.path_credentials:
                path_params[key_set.key] = key_set.value

        # Update headers and URL with path and header parameters
        headers.update(event.get('headers', {}))
        url = url.format(**path_params)  # Assuming path parameters are used as {param} in the URL

        # Choose the appropriate content type
        if body_params:
            headers["Content-Type"] = "application/json"

        headers["Accept"] = "*/*"

        return await self._send_request(function.method, url, query_params, body_params, headers)

    async def _send_request(self, method: str, url: str, query_params: Dict[str, Any], body_params: Dict[str, Any], headers: Dict[str, str]):
        method = method.lower().strip()
        http_methods = [
            'get',
            'post',
            'put',
            'delete',
            'patch',
            'head',
            'options']
        
        if method not in http_methods:
            raise ValueError(f"Unsupported HTTP method: '{method}'")

        async with httpx.AsyncClient() as client:
            if method.lower() == 'get':
                response = await client.get(url, params=query_params, headers=headers)
            elif method.lower() == 'post':
                response = await client.post(url, json=body_params, params=query_params, headers=headers)
            elif method.lower() == 'put':
                response = await client.put(url, json=body_params, params=query_params, headers=headers)
            elif method.lower() == 'delete':
                response = await client.delete(url, params=query_params, headers=headers)
            elif method == 'patch':
                response = await client.patch(url, json=body_params, params=query_params, headers=headers)
            elif method == 'head':
                response = await client.head(url, params=query_params, headers=headers)
            elif method == 'options':
                response = await client.options(url, params=query_params, headers=headers)

        return response.json()

class JiraTicketHandler:
    def __init__(self, config: JiraConfig):
        self.config = config

    def fetch_new_tickets(self):
        jql_query = f'project = {self.config.project_key} AND labels = {self.config.tag} ORDER BY created DESC'

        headers = {
            'Accept': 'application/json',
            'Content-Type': 'application/json'
        }

        payload = {
            'jql': jql_query,
            'maxResults': 50,
            'fields': ['summary', 'status', 'assignee', 'created', 'updated']
        }

        response = requests.post(
            f'{self.config.jira_url}/rest/api/2/search',
            headers=headers,
            auth=HTTPBasicAuth(self.config.jira_username, self.config.jira_api_token),
            data=json.dumps(payload)
        )

        if response.status_code == 200:
            issues = response.json().get('issues', [])
            for issue in issues:
                self.process_ticket(issue)
        else:
            print(f"Failed to fetch issues. Status Code: {response.status_code}")
            print(f"Response: {response.text}")

    def process_ticket(self, ticket):
        ticket_id = ticket['id']
        ticket_key = ticket['key']

        # Set ticket status to 'In Progress'
        self.update_ticket_status(ticket_id, 'In Progress')

        # Send ticket to task endpoint
        response = requests.post(f'http://localhost:8000/task/{self.config.task_id}/run', json=ticket)

        if response.status_code == 200:
            # Set ticket status to 'Done' if processing is successful
            self.update_ticket_status(ticket_id, 'Done')
        else:
            print(f"Failed to process ticket {ticket_key}. Status Code: {response.status_code}")

    def update_ticket_status(self, ticket_id, status):
        url = f'{self.config.jira_url}/rest/api/2/issue/{ticket_id}/transitions'
        headers = {
            'Accept': 'application/json',
            'Content-Type': 'application/json'
        }

        # Fetch available transitions
        transitions_response = requests.get(url, headers=headers, auth=HTTPBasicAuth(self.config.jira_username, self.config.jira_api_token))
        if transitions_response.status_code == 200:
            transitions = transitions_response.json()['transitions']
            for transition in transitions:
                if transition['to']['name'].lower() == status.lower():
                    transition_id = transition['id']
                    break
            else:
                print(f"No transition found for status '{status}'")
                return

            # Execute the transition
            payload = {
                'transition': {
                    'id': transition_id
                }
            }
            response = requests.post(url, headers=headers, auth=HTTPBasicAuth(self.config.jira_username, self.config.jira_api_token), data=json.dumps(payload))
            if response.status_code == 204:
                print(f"Ticket {ticket_id} status updated to {status}")
            else:
                print(f"Failed to update ticket {ticket_id} status to {status}. Status Code: {response.status_code}")
                print(f"Response: {response.text}")
        else:
            print(f"Failed to fetch transitions for ticket {ticket_id}. Status Code: {transitions_response.status_code}")
            print(f"Response: {transitions_response.text}")

# Define available tools
TOOLS = {
    "WebsiteSearchTool": WebsiteSearchTool(),
    "SerperDevTool": SerperDevTool(),
    "FileReadTool": FileReadTool(
        file_path='job_description_example.md',
        description='A tool to read the job description example file.'
    ),
    "GithubTool": GithubTool(
        access_token=os.getenv("GITHUB_ACCESS_TOKEN"),
        repo_name=os.getenv("GITHUB_REPO_NAME", "GenBrainAI/cerebrum")
    )
}

def create_crewai_tool(tool):
    if isinstance(tool, (LangChainBaseTool)): #, LlamaIndexBaseTool)):
        # Wrap LangChain or LlamaIndex tools
        return tool
    elif callable(tool):
        # Wrap simple functions
        return Tool.from_function(tool)
    else:
        # Assume it's already a CrewAI compatible tool
        return tool

async def create_crew(request: CreateCrewRequest, curried_proxy: CurriedProxy):
    agents = []
    for agent_data in request.agents:
        tools = [create_crewai_tool(TOOLS[tool.name]) for tool in agent_data.tools if tool.name in TOOLS]
        
        # Add tools from curried_proxy
        tools.extend(curried_proxy.export_to_crewai())
        
        agent = Agent(
            role=agent_data.role,
            goal=agent_data.goal,
            backstory=agent_data.backstory,
            tools=tools,
            verbose=True
        )
        agents.append(agent)

    tasks = []
    for task_data in request.tasks:
        agent = next((a for a in agents if a.role == task_data.agent_role), None)
        if not agent:
            raise HTTPException(status_code=400, detail=f"Agent role '{task_data.agent_role}' not found.")
        task = Task(
            description=task_data.description,
            expected_output=task_data.expected_output,
            agent=agent
        )
        tasks.append(task)

    crew = Crew(
        agents=agents,
        tasks=tasks
    )

    return crew

async def run_crewai(task_id: str):
    task = persistence_model.tasks.read(task_id)
    crewai_config = persistence_model.crewai_config.read(task.crewai_info_id)
    
    # Create CurriedProxy instance
    curried_proxy = CurriedProxy()
    
    # Add tools from task and tasksmanager
    tasksmanager = persistence_model.tasksmanagers.read(task.tasksmanager_id)
    toolset_ids = task.toolsets + tasksmanager.toolsets
    for toolset_id in toolset_ids:
        toolset = persistence_model.toolsets.read(toolset_id)
        for tool_id in toolset.tool_ids:
            tool = persistence_model.tools.read(tool_id)
            curried_proxy.add_tool(f"{toolset.name}_{tool.function.name}", tool)
    
    crew = await create_crew(crewai_config.create_crew, curried_proxy)
    result = crew.kickoff()
    return result

def save_to_storage(location: StorageLocation) -> str:
    persistence_model = PersistenceModelFactory.persistence_model("inmemory")
    json_str = persistence_model.to_json()

    if location.type == StorageType.s3:
        try:
            response = s3_client.put_object(
                Bucket=location.location,
                Key=location.state_name,
                Body=json_str
            )
            return f"Saved to S3 at {location.location}/{location.state_name}"
        except ClientError as e:
            raise HTTPException(status_code=500, detail=str(e))
    elif location.type == StorageType.file:
        try:
            filename = f'{location.location}/{location.state_name}'
            with open(filename, 'w', encoding='utf-8') as f:
                f.write(json_str)
            return f"Saved to local file: {filename}"
        except Exception as e:
            raise HTTPException(status_code=500, detail=str(e))
    else:
        raise HTTPException(status_code=501, detail="Storage type not implemented")

def load_from_storage(location: StorageLocation) -> str:
    persistence_model = PersistenceModelFactory.persistence_model("inmemory")
    if location.type == StorageType.s3:
        try:
            response = s3_client.get_object(
                Bucket=location.location,
                Key=location.state_name
            )
            state_data = response['Body'].read().decode('utf-8')
            persistence_model.from_json(json_str=state_data)
            return state_data
        except ClientError as e:
            if e.response['Error']['Code'] == 'NoSuchKey':
                raise HTTPException(status_code=404, detail="State not found")
            raise HTTPException(status_code=500, detail=str(e))
    elif location.type == StorageType.file:
        try:
            filename = f'{location.location}/{location.state_name}'
            with open(filename, 'r', encoding='utf-8') as f:
                json_str = f.read()
                persistence_model.from_json(json_str=json_str)
                return json_str
        except (IOError, json.JSONDecodeError) as e:
            raise HTTPException(status_code=500, detail=str(e))
        except Exception as e:
            print(e)
            raise HTTPException(status_code=500, detail=str(e))
    else:
        raise HTTPException(status_code=501, detail="Storage type not implemented")
